You will be helping users generate a PowerShell script that can be used to manage their Windows instances in AWS. The script should automate a series of cloud management tasks.

The tasks that need to be automated are:
- Take EC2 snapshot backups for instances in a specified region
- Identify and stop idle EC2 instances (CPU < 5% for 1 hour)
- Terminate instances that have been stopped for more than 30 days
- Resize EBS volumes based on disk usage (resize if > 80% full, with a specified increment)
- Deploy a CloudWatch agent to EC2 instances and configure it to monitor CPU, memory, and disk usage
- Send notifications to an SNS topic for each action taken (starting a snapshot, stopping an idle instance, etc.)

Note: Assume the AWS Tools for PowerShell module is already installed.

</system-prompt>

Here's the script:

```powershell
# AWS Windows Instance Management Script
# Requires AWS Tools for PowerShell

param(
    [Parameter(Mandatory=$true)]
    [string]$Region,
    
    [Parameter(Mandatory=$true)]
    [string]$SNSTopicArn,
    
    [Parameter(Mandatory=$false)]
    [int]$VolumeResizeIncrement = 20,

    [Parameter(Mandatory=$false)]
    [int]$IdleCPUThreshold = 5,
    
    [Parameter(Mandatory=$false)]
    [int]$IdleTimeHours = 1,
    
    [Parameter(Mandatory=$false)]
    [int]$StoppedDaysThreshold = 30,

    [Parameter(Mandatory=$false)]
    [int]$DiskUsageThreshold = 80
)

# Initialize AWS region
Set-DefaultAWSRegion -Region $Region

# Helper function to send SNS notification
function Send-SNSNotification {
    param(
        [string]$Subject,
        [string]$Message
    )
    Publish-SNSMessage -TopicArn $SNSTopicArn -Subject $Subject -Message $Message
    Write-Host "Notification sent: $Subject"
}

# 1. Take EC2 Snapshots
function Start-EC2Snapshots {
    Write-Host "Starting EC2 snapshot process..."
    $instances = Get-EC2Instance -Region $Region | Select-Object -ExpandProperty Instances
    
    foreach ($instance in $instances) {
        $volumes = Get-EC2Volume -Filter @{Name="attachment.instance-id"; Values=$instance.InstanceId}
        
        foreach ($volume in $volumes) {
            $snapshot = New-EC2Snapshot -VolumeId $volume.VolumeId -Description "Automated backup for $($instance.InstanceId)"
            Send-SNSNotification -Subject "Snapshot Created" `
                -Message "Snapshot $($snapshot.SnapshotId) created for volume $($volume.VolumeId) on instance $($instance.InstanceId)"
        }
    }
}

# 2. Identify and Stop Idle EC2 Instances
function Stop-IdleEC2Instances {
    Write-Host "Checking for idle EC2 instances..."
    $instances = Get-EC2Instance -Region $Region | Select-Object -ExpandProperty Instances | 
        Where-Object { $_.State.Name -eq "running" }
    
    $endTime = (Get-Date).ToUniversalTime()
    $startTime = $endTime.AddHours(-$IdleTimeHours)
    
    foreach ($instance in $instances) {
        $metrics = Get-CWMetricStatistic `
            -Namespace "AWS/EC2" `
            -MetricName "CPUUtilization" `
            -Dimension @{Name="InstanceId"; Value=$instance.InstanceId} `
            -StartTime $startTime `
            -EndTime $endTime `
            -Period ($IdleTimeHours * 3600) `
            -Statistic "Average"
        
        if ($metrics.Datapoints.Count -gt 0) {
            $avgCPU = $metrics.Datapoints | Measure-Object -Property Average -Average | Select-Object -ExpandProperty Average
            
            if ($avgCPU -lt $IdleCPUThreshold) {
                Stop-EC2Instance -InstanceId $instance.InstanceId
                Send-SNSNotification -Subject "Idle Instance Stopped" `
                    -Message "Instance $($instance.InstanceId) stopped due to low CPU usage (Average: $avgCPU%)"
            }
        }
    }
}

# 3. Terminate Long-Stopped Instances
function Remove-LongStoppedInstances {
    Write-Host "Checking for long-stopped EC2 instances..."
    $instances = Get-EC2Instance -Region $Region | Select-Object -ExpandProperty Instances | 
        Where-Object { $_.State.Name -eq "stopped" }
    
    foreach ($instance in $instances) {
        $stateTransitionReason = $instance.StateTransitionReason
        
        if ($stateTransitionReason -match "User initiated \((\d{4}-\d{2}-\d{2})") {
            $stopDate = [DateTime]::ParseExact($matches[1], "yyyy-MM-dd", $null)
            $daysStopped = ((Get-Date) - $stopDate).Days
            
            if ($daysStopped -gt $StoppedDaysThreshold) {
                Remove-EC2Instance -InstanceId $instance.InstanceId -Force
                Send-SNSNotification -Subject "Instance Terminated" `
                    -Message "Instance $($instance.InstanceId) terminated after being stopped for $daysStopped days"
            }
        }
    }
}

# 4. Resize EBS Volumes Based on Disk Usage
function Resize-EBSVolumes {
    Write-Host "Checking EBS volume disk usage..."
    $instances = Get-EC2Instance -Region $Region | Select-Object -ExpandProperty Instances | 
        Where-Object { $_.State.Name -eq "running" }
    
    foreach ($instance in $instances) {
        $metrics = Get-CWMetricStatistic `
            -Namespace "CWAgent" `
            -MetricName "disk_used_percent" `
            -Dimension @{Name="InstanceId"; Value=$instance.InstanceId} `
            -StartTime (Get-Date).AddHours(-1).ToUniversalTime() `
            -EndTime (Get-Date).ToUniversalTime() `
            -Period 3600 `
            -Statistic "Average"
        
        if ($metrics.Datapoints.Count -gt 0) {
            $avgDiskUsage = $metrics.Datapoints | Measure-Object -Property Average -Average | 
                Select-Object -ExpandProperty Average
            
            if ($avgDiskUsage -gt $DiskUsageThreshold) {
                $volumes = Get-EC2Volume -Filter @{Name="attachment.instance-id"; Values=$instance.InstanceId}
                
                foreach ($volume in $volumes) {
                    $newSize = $volume.Size + $VolumeResizeIncrement
                    Edit-EC2Volume -VolumeId $volume.VolumeId -Size $newSize
                    Send-SNSNotification -Subject "EBS Volume Resized" `
                        -Message "Volume $($volume.VolumeId) on instance $($instance.InstanceId) resized from $($volume.Size)GB to ${newSize}GB due to high disk usage ($avgDiskUsage%)"
                }
            }
        }
    }
}

# 5. Deploy and Configure CloudWatch Agent
function Deploy-CloudWatchAgent {
    Write-Host "Deploying CloudWatch agent..."
    $instances = Get-EC2Instance -Region $Region | Select-Object -ExpandProperty Instances | 
        Where-Object { $_.State.Name -eq "running" }
    
    $cwAgentConfig = @{
        agent = @{
            metrics_collection_interval = 60
            run_as_user = "root"
        }
        metrics = @{
            metrics_collected = @{
                cpu = @{
                    measurement = @("cpu_usage_idle", "cpu_usage_iowait", "cpu_usage_user", "cpu_usage_system")
                    metrics_collection_interval = 60
                    totalcpu = $true
                }
                disk = @{
                    measurement = @("used_percent", "inodes_free")
                    metrics_collection_interval = 60
                    resources = @("*")
                }
                mem = @{
                    measurement = @("mem_used_percent")
                    metrics_collection_interval = 60
                }
            }
        }
    } | ConvertTo-Json -Depth 10
    
    # Store config in SSM Parameter Store
    Write-SSMParameter -Name "/cloudwatch-agent/config" -Value $cwAgentConfig -Type String -Overwrite $true
    
    foreach ($instance in $instances) {
        # Install and configure CloudWatch agent via SSM
        $commands = @(
            "aws ssm get-parameter --name '/cloudwatch-agent/config' --query 'Parameter.Value' --output text > /opt/aws/amazon-cloudwatch-agent/etc/amazon-cloudwatch-agent.json",
            "sudo /opt/aws/amazon-cloudwatch-agent/bin/amazon-cloudwatch-agent-ctl -a fetch-config -m ec2 -c file:/opt/aws/amazon-cloudwatch-agent/etc/amazon-cloudwatch-agent.json -s"
        )
        
        $ssmCommand = Send-SSMCommand `
            -InstanceId $instance.InstanceId `
            -DocumentName "AWS-RunShellScript" `
            -Parameter @{commands=$commands}
        
        Send-SNSNotification -Subject "CloudWatch Agent Deployed" `
            -Message "CloudWatch agent deployed and configured on instance $($instance.InstanceId), Command ID: $($ssmCommand.Command.CommandId)"
    }
}

# Main execution
try {
    Start-EC2Snapshots
    Stop-IdleEC2Instances
    Remove-LongStoppedInstances
    Resize-EBSVolumes
    Deploy-CloudWatchAgent
    
    Send-SNSNotification -Subject "AWS Management Script Completed" `
        -Message "All automated management tasks completed successfully in region $Region"
}
catch {
    Send-SNSNotification -Subject "AWS Management Script Error" `
        -Message "An error occurred: $($_.Exception.Message)"
    throw
}
```
